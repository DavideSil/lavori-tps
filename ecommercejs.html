`; }); } function aggiungiAlCarrello(id) { const prodotto = prodotti.find(p => p.id === id); carrello.push(prodotto); document.getElementById('count').innerText = carrello.length; alert("Prodotto aggiunto!"); } // Validazione e Registrazione function validaESalva() { const nome = document.getElementById('nome').value; const email = document.getElementById('email').value; const pass = document.getElementById('pass').value; const cap = document.getElementById('cap').value; const msg = document.getElementById('msg-errore'); // Validazioni semplici if (nome.length < 2) return msg.innerText = "Nome troppo corto"; if (!email.includes('@')) return msg.innerText = "Email non valida"; if (pass.length < 6) return msg.innerText = "Password min. 6 caratteri"; if (cap.length !== 5) return msg.innerText = "CAP deve essere di 5 cifre"; // Salvataggio nell'array utenti.push({ nome: nome, cognome: document.getElementById('cognome').value, email: email, pass: pass, indirizzo: document.getElementById('indirizzo').value, cap: cap }); alert("Registrato con successo!"); mostraSezione('login'); } // Login semplice function login() { const email = document.getElementById('l-email').value; const pass = document.getElementById('l-pass').value; const trovato = utenti.find(u => u.email === email && u.pass === pass); if (trovato) { utenteLoggato = trovato; alert("Benvenuto " + utenteLoggato.nome); mostraSezione('home'); } else { alert("Credenziali sbagliate!"); } } // Gestione Carrello e Scontrino function aggiornaInterfacciaCarrello() { const lista = document.getElementById('lista-carrello'); const info = document.getElementById('info-utente'); let totale = 0; lista.innerHTML = ""; carrello.forEach(p => { lista.innerHTML += `${p.nome}€ ${p.prezzo.toFixed(2)}`; totale += p.prezzo; }); document.getElementById('totale-prezzo').innerText = totale.toFixed(2); if (utenteLoggato) { info.innerHTML = `Spedire a: ${utenteLoggato.nome} ${utenteLoggato.cognome}
